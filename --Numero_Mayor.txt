Numero Mayor

-- Creamos el procedimiento
CREATE OR REPLACE PROCEDURE NumeroMayor AS

    -- Declaramos las variables
    numero1 NUMBER;
    numero2 NUMBER;
    numero3 NUMBER;
    mayor NUMBER;

BEGIN

    -- Asignamos valores a las variables
    numero1 := 25;
    numero2 := 50;
    numero3 := 35;

    -- Comparamos los tres números
    IF numero1 >= numero2 AND numero1 >= numero3 THEN

        mayor := numero1;

    ELSIF numero2 >= numero1 AND numero2 >= numero3 THEN

        mayor := numero2;

    ELSE

        mayor := numero3;

    END IF;

    -- Mostramos los resultados
    DBMS_OUTPUT.PUT_LINE('Numero 1: ' || numero1);
    DBMS_OUTPUT.PUT_LINE('Numero 2: ' || numero2);
    DBMS_OUTPUT.PUT_LINE('Numero 3: ' || numero3);
    DBMS_OUTPUT.PUT_LINE('El numero mayor es: ' || mayor);

END;
/

-- Ejecutamos el procedimiento
BEGIN
    NumeroMayor;
END;
/
